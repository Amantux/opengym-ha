#include <tunables/global>

# openGym add-on AppArmor profile. Not declared in config.yaml: the Supervisor finds this
# file next to it and matches the `profile` name below. It requires exactly one line that
# starts with `profile` and renames only that one, so the child profile further down is
# indented (and is then <add-on slug>//app).
#
# Runtime: the official openGym api image (node:22-alpine, busybox) plus nginx and gosu, no
# s6-overlay (init: false). /usr/local/bin/opengym-entrypoint.sh is PID 1 and stays root: it
# chowns /data, creates /run/opengym, renders nginx's config, copies the bundled integration
# into Home Assistant's configuration folder when install_integration is on, starts three
# servers as the unprivileged `app` user through gosu, and supervises them. The servers are
# the openGym api (node, 127.0.0.1:3000), nginx (127.0.0.1:8080, the app's static files and
# /api proxy) and our gateway (node, :8099, ingress sign-in and the Home Assistant
# integration API). Supervisor discovery (node) runs as root, so the Supervisor token never
# sits in an environment uid 1000 can read. The gateway reaches the api over a unix socket in
# /run/opengym and the Supervisor API over HTTP. Data lives in /data.
#
# Two profiles:
# - `opengym` (this one): the entrypoint and the busybox/gosu helpers it runs, as root.
# - `opengym//app` (the child below): every node and nginx process, whoever starts them and
#   whatever user they run as: the api, the gateway, its CLI (discovery too) and nginx, and
#   anything they run in turn. It is the same as this profile without the capabilities, and it
#   can never touch /homeassistant (Home Assistant's configuration folder, mounted read-write
#   for install_integration). So a bug in openGym or the gateway, even one that ran code as
#   root, cannot read secrets.yaml or change Home Assistant's configuration.
#
# Strategy: allow the app's own file I/O (broad `file` rule) and let AppArmor's default-deny
# confine everything else: every capability, mount, module load, ptrace and raw socket NOT
# granted below is denied. Paths can be tightened once validated under enforcement on a real
# Home Assistant install.
profile opengym flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Code in /app and /opt/gateway, data in /data, nginx's files, /run/opengym, /tmp, /proc,
  # and /homeassistant for the integration installer (opengym-install-integration.sh).
  file,

  # The entrypoint (PID 1) forwards TERM to every server and polls them with kill -0.
  signal,

  # Boot-time ownership work as root (chown -R /data, chmod), gosu's switch to `app`
  # (setuid/setgid: without them gosu fails and the add-on dies before any log line a user
  # would recognise), and root signalling processes that belong to `app` (kill).
  capability setuid,
  capability setgid,
  capability dac_override,
  capability chown,
  capability fowner,
  capability fsetid,
  capability kill,

  # Execute within THIS profile (ix = inherit): the entrypoint's busybox (sh, cp, mv, sleep,
  # kill, ...) in /bin, gosu/envsubst/env/wget in /usr/bin.
  /usr/local/bin/** ix,
  /usr/sbin/** ix,
  /usr/bin/** ix,
  /bin/** ix,
  # ...except node and nginx, which always start in the child profile. An exact path beats
  # the globs above. cx, not Cx: the servers need the environment the entrypoint gives them.
  /usr/local/bin/node cx -> app,
  /usr/sbin/nginx cx -> app,

  # TCP: the three servers, the Supervisor API, and the AI Coach's HTTPS providers. UDP and
  # netlink: DNS resolution (the resolver enumerates interfaces over netlink, and without it
  # name lookups fail silently). unix stream: the api's internal socket.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network netlink raw,
  network unix stream,

  profile app flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    # Everything the parent allows, except Home Assistant's configuration folder. An explicit
    # deny always wins over an allow, whatever runs here (node's child processes too).
    file,
    deny /homeassistant/ rwklmx,
    deny /homeassistant/** rwklmx,

    # nginx's master signals its worker; the entrypoint stops everything with TERM.
    signal,

    # No capability at all: the servers run as `app`, and discovery, the one root process
    # here, only reads root-owned files and talks HTTP.

    /usr/local/bin/** ix,
    /usr/sbin/** ix,
    /usr/bin/** ix,
    /bin/** ix,

    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network netlink raw,
    network unix stream,
  }
}
